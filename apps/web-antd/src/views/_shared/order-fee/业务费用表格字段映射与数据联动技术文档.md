```
async function onChange(newVal: any) {
  row[column.field] = newVal;

  if (newVal) {
    // 获取汇率详情
    const exchangeRateData = await getExchangeRateDetail(newVal);

    // 判断是否为本位币
    const isLocalCurrency = await checkIfLocalCurrency(newVal);

    row['exchangeRate'] = undefined;
    row['__isLocalCurrency'] = false;
  }
}
```

```
``typescript
async function fillQuantityByUnit(row: any, unitName: string) {
  const orderDetail = await loadOrderDetailCached(row['transportOrderId']);
  const transportOrder = orderDetail.transportOrder;
  const unitNameLower = unitName.toLowerCase();

  if (unitNameLower === '票' || unitNameLower === 'order') {
    row['quantity'] = 1;
  }
  else if (unitNameLower === '毛重' || unitNameLower === 'kgs') {
    row['quantity'] = transportOrder.kgs || 0;
  }
  else if (unitNameLower === '尺码' || unitNameLower === 'cbm') {
    row['quantity'] = transportOrder.cbm || 0;
  }
  else if (unitNameLower === '件数' || unitNameLower === 'pkgs') {
    row['quantity'] = transportOrder.pkgs || 0;
  }
  else if (unitNameLower === 'teu') {
    // 累加所有箱型的TEU
    let totalTeu = 0;
    for (const ctn of transportOrder.orderCtns) {
      const ctnDetail = await loadCtnCodeCached(ctn.ctnCodeId);
      if (ctnDetail?.teu !== undefined) {
        totalTeu += ctnDetail.teu;
      }
    }
    row['quantity'] = totalTeu;
  }
  else {
    // 箱型：统计该箱型的数量
    row['quantity'] = transportOrder.orderCtns.filter(
      (ctn) => ctn.ctnCodeName === unitName
    ).length;
  }
}
```

```
## 五、缓存机制

### 5.1 订单详情缓存

**位置**：`vxe-table.ts`

``typescript
// 全局缓存
const orderDetailCache = new Map<string, any>();
const orderDetailLoading = new Map<string, Promise<any>>();

async function loadOrderDetailCached(transportOrderId: string) {
  // 1. 检查缓存
  if (orderDetailCache.has(transportOrderId)) {
    return orderDetailCache.get(transportOrderId);
  }

  // 2. 防止并发请求
  if (orderDetailLoading.has(transportOrderId)) {
    return await orderDetailLoading.get(transportOrderId);
  }

  // 3. 发起请求并缓存
  const loadingPromise = getSeaExportDetail(transportOrderId)
    .then((detail) => {
      if (detail) {
        orderDetailCache.set(transportOrderId, detail);
      }
      orderDetailLoading.delete(transportOrderId);
      return detail;
    })
    .catch((error) => {
      orderDetailLoading.delete(transportOrderId);
      throw error;
    });

  orderDetailLoading.set(transportOrderId, loadingPromise);
  return await loadingPromise;
}
```

```
## 八、注意事项

### 8.1 性能优化
1. **缓存策略**：订单详情和集装箱详情使用全局缓存，避免重复API调用
2. **防并发**：使用 `Promise` 缓存防止同一资源多次并发请求
3. **懒加载**：仅在需要时触发联动逻辑

### 8.2 边界处理
1. **空值处理**：所有联动逻辑都包含空值检查
2. **除零保护**：金额计算时检查数量是否为0
3. **精度控制**：单价保留4位小数，金额保留2位小数

### 8.3 编辑权限
``typescript
const canEditFee = (feeStatus: number): boolean => {
  const editableStatuses = [
    0, // 录入中
    3, // 审核驳回
    4, // 申请修改
    5, // 申请删除
  ];
  return editableStatuses.includes(feeStatus);
};
```

所有可编辑字段的 `disabled` 属性都基于此函数判断。

### 8.4 已知问题

1. TypeScript 警告：部分解构参数未使用（原有代码问题）
2. 隐式 any 类型：部分回调函数缺少类型定义
3. 这些警告不影响功能正常运行

```
### 12.3 实现的联动功能

#### ✅ 费用代码联动 (`handleFeeCodeChange`)
- 自动填充行业类别（根据收付类型选择 defaultDebitName/defaultCreditName）
- 自动填充结算对象（通过 `fillSettlementIdByIndustryCategory`）
- 自动填充币别
- 自动判断本位币并设置汇率（本位币=1，非本位币使用 drValue/crValue）
- 自动填充税率
- 自动填充单位
- 根据单位类型自动填充数量（箱型/票/毛重/尺码/件数/TEU）

#### ✅ 行业类别联动 (`handleIndustryCategoryChange`)
- 将 key 转换为 value（字母代码）
- 自动填充对应的结算对象

#### ✅ 币别联动 (`handleCurrencyChange`)
- 获取汇率详情
- 判断是否为本位币
- 本位币：汇率=1，禁用编辑
- 非本位币：根据收付类型选择 drValue/crValue

#### ✅ 单位联动 (`handleUnitChange`)
- 票/ORDER：数量=1
- 毛重/KGS：从订单 kgs 字段获取
- 尺码/CBM：从订单 cbm 字段获取
- 件数/PKGS：从订单 pkgs 字段获取
- TEU：累加所有箱型的 teu 值
- 箱型：统计该箱型的数量

#### ✅ 数值计算联动
- **含税单价变化**：计算含税金额、不含税单价、不含税金额
- **数量变化**：计算含税金额、不含税金额
- **税率变化**：重新计算不含税单价、不含税金额
- **含税金额变化**：反向计算含税单价、不含税单价、不含税金额
```
